## The measurement script of the serial runs
```
#!/usr/bin/env bash
# Measurement only: the commands of .config/gates.sh, one after the other.
cd /data/dev/xumiga/lpg; t=/dev/shm/lpg-target; logs=$1; mkdir -p $logs
g() { local n=$1 s=$SECONDS; shift; "$@" >"$logs/$n.log" 2>&1; echo "$n: exit $?, $((SECONDS-s)) s"; }
start=$SECONDS
g fmt cargo fmt --all -- --check
g clippy cargo clippy --all-targets --target-dir "$t/clippy" -- -D warnings
g cfgfuzz env RUSTFLAGS="--cfg fuzzing" cargo clippy --lib --target-dir "$t/cfgfuzz" -- -D warnings
g nextest cargo nextest run -p lpg --no-fail-fast
g loom env RUSTFLAGS="--cfg loom" cargo test -p lpg --lib --release --target-dir "$t/loom" loom_
g shuttle env RUSTFLAGS="--cfg shuttle" cargo test -p lpg --lib --target-dir "$t/shuttle" shuttle_
echo "without stress: $((SECONDS-start)) s"
du -sh $t/cfgfuzz $t/loom $t/shuttle
rm -rf "$t/cfgfuzz" "$t/loom" "$t/shuttle"
g stress cargo nextest run -p lpg --profile stress --no-fail-fast
echo "all: $((SECONDS-start)) s"
```
